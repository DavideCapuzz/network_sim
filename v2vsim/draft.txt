class ExternalMobility : public MovingMobilityBase
{
  private:

  public:
    ExternalMobility();
    virtual ~ExternalMobility();

// 5. ExternalMobility Implementation (ExternalMobility.cc)

#include "ExternalMobility.h"

Define_Module(ExternalMobility);

ExternalMobility::ExternalMobility()
{
    updateTimer = nullptr;
    sockfd = -1;
}

ExternalMobility::~ExternalMobility()
{
    cancelAndDelete(updateTimer);
    if (sockfd >= 0)
        close(sockfd);
}

void ExternalMobility::initialize(int stage)
{
    MovingMobilityBase::initialize(stage);

    if (stage == INITSTAGE_LOCAL) {
        externalHost = par("externalHost");
        externalPort = par("externalPort");
        updateInterval = par("updateInterval");

        lastPosition.x = par("initialX");
        lastPosition.y = par("initialY");
        lastPosition.z = par("initialZ");

        updateTimer = new cMessage("updateTimer");
    }
    else if (stage == INITSTAGE_PHYSICAL_ENVIRONMENT) {
        connectToExternalSimulator();
        scheduleAt(simTime() + updateInterval, updateTimer);
    }
}

void ExternalMobility::connectToExternalSimulator()
{
    sockfd = socket(AF_INET, SOCK_DGRAM, 0);
    if (sockfd < 0) {
        EV_ERROR << "Failed to create socket" << endl;
        return;
    }

    memset(&servaddr, 0, sizeof(servaddr));
    servaddr.sin_family = AF_INET;
    servaddr.sin_port = htons(externalPort);
    inet_pton(AF_INET, externalHost, &servaddr.sin_addr);

    EV_INFO << "Connected to external simulator at " << externalHost
            << ":" << externalPort << endl;
}

void ExternalMobility::handleSelfMessage(cMessage *msg)
{
    if (msg == updateTimer) {
        requestPositionUpdate();

        double x, y, z;
        if (receivePositionUpdate(x, y, z)) {
            targetPosition.x = x;
            targetPosition.y = y;
            targetPosition.z = z;

            move();
        }

        scheduleAt(simTime() + updateInterval, updateTimer);
    }
}

void ExternalMobility::requestPositionUpdate()
{
    // Request format: "GET_POS:<module_id>"
    char buffer[256];
    snprintf(buffer, sizeof(buffer), "GET_POS:%s", getFullPath().c_str());

    sendto(sockfd, buffer, strlen(buffer), 0,
           (struct sockaddr*)&servaddr, sizeof(servaddr));
}

bool ExternalMobility::receivePositionUpdate(double &x, double &y, double &z)
{
    char buffer[256];
    socklen_t len = sizeof(servaddr);

    // Set timeout for receiving
    struct timeval tv;
    tv.tv_sec = 0;
    tv.tv_usec = 100000; // 100ms timeout
    setsockopt(sockfd, SOL_SOCKET, SO_RCVTIMEO, &tv, sizeof(tv));

    int n = recvfrom(sockfd, buffer, sizeof(buffer), 0,
                     (struct sockaddr*)&servaddr, &len);

    if (n > 0) {
        buffer[n] = '\0';
        // Expected format: "POS:x,y,z"
        if (sscanf(buffer, "POS:%lf,%lf,%lf", &x, &y, &z) == 3) {
            return true;
        }
    }

    return false;
}

void ExternalMobility::setInitialPosition()
{
    lastPosition.x = par("initialX");
    lastPosition.y = par("initialY");
    lastPosition.z = par("initialZ");
}

void ExternalMobility::move()
{
    lastPosition = targetPosition;
    emitMobilityStateChangedSignal();
}

void ExternalMobility::orient()
{
    // No orientation change needed
}

// 10. External Simulator Interface (Python Example)

"""
External Simulator Interface - Python Example
This script demonstrates how to provide position updates to OMNeT++
"""

import socket
import json
import time
import math

class ExternalSimulator:
    def __init__(self, host='localhost', port=9999):
        self.host = host
        self.port = port
        self.sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
        self.sock.bind((host, port))

        # Robot positions (simulated)
        self.robots = {
            'robot[0]': {'x': 100, 'y': 100, 'z': 0, 'vx': 1, 'vy': 0.5},
            'robot[1]': {'x': 300, 'y': 150, 'z': 0, 'vx': -0.5, 'vy': 1},
            'robot[2]': {'x': 200, 'y': 250, 'z': 0, 'vx': 0.7, 'vy': -0.7},
        }

        self.time = 0

    def update_positions(self, dt):
        """Update robot positions based on simple motion model"""
        self.time += dt

        for robot_id, pos in self.robots.items():
            # Simple circular motion with drift
            pos['x'] += pos['vx'] * dt + math.sin(self.time * 0.1) * 5
            pos['y'] += pos['vy'] * dt + math.cos(self.time * 0.1) * 5

            # Boundary check (keep in 1000x1000 area)
            if pos['x'] < 0 or pos['x'] > 1000:
                pos['vx'] *= -1
            if pos['y'] < 0 or pos['y'] > 1000:
                pos['vy'] *= -1

    def handle_request(self):
        """Handle position requests from OMNeT++"""
        try:
            data, addr = self.sock.recvfrom(1024)
            request = data.decode('utf-8')

            if request.startswith('GET_POS:'):
                robot_id = request.split(':')[1]

                # Find matching robot
                for rid, pos in self.robots.items():
                    if rid in robot_id:
                        response = f"POS:{pos['x']},{pos['y']},{pos['z']}"
                        self.sock.sendto(response.encode(), addr)
                        print(f"Sent position for {rid}: {response}")
                        break

        except socket.timeout:
            pass

    def run(self):
        """Main simulation loop"""
        print(f"External Simulator running on {self.host}:{self.port}")
        self.sock.settimeout(0.01)  # 10ms timeout

        last_update = time.time()

        try:
            while True:
                current_time = time.time()
                dt = current_time - last_update

                if dt >= 0.1:  # Update every 100ms
                    self.update_positions(dt)
                    last_update = current_time

                self.handle_request()

        except KeyboardInterrupt:
            print("\\nShutting down...")
            self.sock.close()

if __name__ == '__main__':
    sim = ExternalSimulator()
    sim.run()
